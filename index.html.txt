    <script>
        const area = document.getElementById("clickArea");
        const cpsDisplay = document.getElementById("cps");
        const clicksDisplay = document.getElementById("clicks");
        const timeDisplay = document.getElementById("time");
        const title = document.getElementById("areaTitle");
        const text = document.getElementById("areaText");
        const startBtn = document.getElementById("startBtn");
        const historyList = document.getElementById("historyList");
        const modeButtons = document.querySelectorAll("[data-mode]");

        let mode = "live";
        let clicks = 0;
        let started = false;
        let finished = false;
        let startTime = 0;
        let elapsed = 0;
        let timer = null;

        let history = JSON.parse(
            localStorage.getItem("cpsHistory") || "[]"
        );

        function renderHistory() {
            if (history.length === 0) {
                historyList.innerHTML =
                    '<p class="muted">Здесь появятся твои последние результаты.</p>';
                return;
            }

            historyList.innerHTML = history.map(item => `
                <div class="history-item">
                    <div>
                        <strong>${item.mode}</strong>
                        <div class="muted">${item.clicks} кликов · ${item.time} сек.</div>
                    </div>
                    <strong class="green">${item.cps} CPS</strong>
                </div>
            `).join("");
        }

        function updateStats() {
            const seconds = started
                ? (performance.now() - startTime) / 1000
                : elapsed;

            const cps = seconds > 0 ? clicks / seconds : 0;

            cpsDisplay.textContent = cps.toFixed(2);
            clicksDisplay.textContent = clicks;
            timeDisplay.textContent = seconds.toFixed(1);
        }

        function stopTest() {
            if (!started || finished) return;

            elapsed = (performance.now() - startTime) / 1000;
            started = false;
            finished = true;
            clearInterval(timer);

            if (mode !== "live") {
                elapsed = Number(mode);
            }

            const cps = elapsed > 0 ? clicks / elapsed : 0;

            title.textContent = "Тест завершён!";
            text.textContent =
                `Результат: ${cps.toFixed(2)} CPS. Нажми «Начать заново».`;

            history.unshift({
                mode: mode === "live" ? "Live" : mode + " сек.",
                clicks: clicks,
                time: elapsed.toFixed(1),
                cps: cps.toFixed(2)
            });

            history = history.slice(0, 3);
            localStorage.setItem("cpsHistory", JSON.stringify(history));
            renderHistory();
            updateStats();
        }

        function beginTest() {
            clicks = 0;
            elapsed = 0;
            started = true;
            finished = false;
            startTime = performance.now();

            title.textContent = "КЛИКАЙ!";
            text.textContent = mode === "live"
                ? "Нажми пробел, чтобы завершить тест"
                : `Тест на ${mode} секунд`;

            clearInterval(timer);

            timer = setInterval(() => {
                updateStats();

                if (mode !== "live" &&
                    (performance.now() - startTime) / 1000 >= Number(mode)) {
                    stopTest();
                }
            }, 50);

            updateStats();
        }

        function resetTest() {
            clearInterval(timer);
            started = false;
            finished = false;
            clicks = 0;
            elapsed = 0;

            cpsDisplay.textContent = "0.00";
            clicksDisplay.textContent = "0";
            timeDisplay.textContent = "0.0";

            title.textContent = "Нажми, чтобы начать!";
            text.textContent = "Каждый клик увеличивает твой результат";
        }

        area.addEventListener("click", () => {
            if (finished) return;

            if (!started) beginTest();

            if (finished) return;

            clicks++;
            updateStats();

            if (mode !== "live" &&
                (performance.now() - startTime) / 1000 >= Number(mode)) {
                stopTest();
            }
        });

        modeButtons.forEach(button => {
            button.addEventListener("click", () => {
                mode = button.dataset.mode;

                modeButtons.forEach(btn =>
                    btn.classList.toggle("active", btn === button)
                );

                resetTest();
            });
        });

        startBtn.addEventListener("click", resetTest);

        document.addEventListener("keydown", event => {
            if (event.code === "Space" &&
                mode === "live" &&
                started &&
                !finished) {
                event.preventDefault();
                stopTest();
            }
        });

        renderHistory();
    </script>
